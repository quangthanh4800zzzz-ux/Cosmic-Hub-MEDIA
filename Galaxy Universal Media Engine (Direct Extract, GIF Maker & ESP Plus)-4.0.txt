        function srgbToOklab(r8, g8, b8) {
            let rl = r8 / 255, gl = g8 / 255, bl = b8 / 255;
            rl = rl > 0.04045 ? Math.pow((rl + 0.055) / 1.055, 2.4) : rl / 12.92;
            gl = gl > 0.04045 ? Math.pow((gl + 0.055) / 1.055, 2.4) : gl / 12.92;
            bl = bl > 0.04045 ? Math.pow((bl + 0.055) / 1.055, 2.4) : bl / 12.92;

            const mp = 0.2119034982 * rl + 0.6806995451 * gl + 0.1073969566 * bl;
            const sp = 0.0883024619 * rl + 0.2817188376 * gl + 0.6299787005 * bl;
            const lc = Math.cbrt(lp), mc = Math.cbrt(mp), sc = Math.cbrt(sp);
            return [
                0.2104542553 * lc + 0.7936177850 * mc - 0.0040720468 * sc,

        function mmcQuantize(rgba, nColors, activityMap, detailBoost) {

                const idx = ((rgba[o] >> SHIFT) * SIDE2) + ((rgba[o + 1] >> SHIFT) * SIDE) + (rgba[o + 2] >> SHIFT);
                histo[idx] += w;
            }

        function indexWithDither(rgba, palette, nColors, width, height, activityMap, quality) {
            const indexed = new Uint8Array(width * height);
            const eR = new Float32Array(width * height);
            const eG = new Float32Array(width * height);
            const eB = new Float32Array(width * height);

            const palLab = new Float32Array(nColors * 3);
            for (let j = 0; j < nColors; j++) {

                palLab[j*3]=lab[0];palLab[j*3+1]=lab[1];palLab[j*3+2]=lab[2];
            }
            const CSIZE = 32*32*32;
            const cache = new Int16Array(CSIZE).fill(-1);

            function nearest(r, g, b) {
                const key = ((r>>3)<<10)|((g>>3)<<5)|(b>>3);
                if (cache[key] >= 0) return cache[key];
                const qL = srgbToOklab(r, g, b);
                let best=0,bd=1e18;
                for (let j=0;j<nColors;j++) {
                    const dL=qL[0]-palLab[j*3],da=qL[1]-palLab[j*3+1],db=qL[2]-palLab[j*3+2];
                    const d=dL*dL+da*da+db*db;
                    if(d<bd){bd=d;best=j;if(d<1e-12)break;}
                }
                return (cache[key]=best);
            }

            const isMaximum = quality === 'maximum';
            const isUltra   = quality === 'ultra';
            const isStandard = quality === 'standard';

            for (let y = 0; y < height; y++) {
                for (let x = 0; x < width; x++) {
                    const pi = y * width + x;
                    const ri = pi << 2;
                    const r = Math.max(0,Math.min(255,Math.round(rgba[ri]   +eR[pi])));
                    const g = Math.max(0,Math.min(255,Math.round(rgba[ri+1] +eG[pi])));
                    const b = Math.max(0,Math.min(255,Math.round(rgba[ri+2] +eB[pi])));
                    const idx = nearest(r, g, b);
                    indexed[pi] = idx;
                    const er = r - palette[idx*3];
                    const eg = g - palette[idx*3+1];
                    const eb = b - palette[idx*3+2];

                    // Adaptive dither weight
                    let dw;
                    if (isMaximum || isUltra) {
                        dw = 1.0;
                    } else if (isStandard) {
                        dw = activityMap ? Math.min(1, activityMap[pi] * 6) : 1.0;
                    } else {
                        dw = activityMap ? Math.min(1, 0.3 + activityMap[pi] * 4) : 1.0;
                    }
                    if (dw < 0.01) continue;

                    if (isMaximum) {
                        // ── Atkinson dithering (6/8 error distribution) ──
                        // Right 1
                        if (x < width-1) { eR[pi+1]+=er*dw/8; eG[pi+1]+=eg*dw/8; eB[pi+1]+=eb*dw/8; }
                        // Right 2
                        if (x < width-2) { eR[pi+2]+=er*dw/8; eG[pi+2]+=eg*dw/8; eB[pi+2]+=eb*dw/8; }
                        if (y < height-1) {
                            if (x > 0)        { eR[pi+width-1]+=er*dw/8; eG[pi+width-1]+=eg*dw/8; eB[pi+width-1]+=eb*dw/8; }
                            eR[pi+width]  +=er*dw/8; eG[pi+width]  +=eg*dw/8; eB[pi+width]  +=eb*dw/8;
                            if (x < width-1)  { eR[pi+width+1]+=er*dw/8; eG[pi+width+1]+=eg*dw/8; eB[pi+width+1]+=eb*dw/8; }
                        }
                        if (y < height-2) { eR[pi+width*2]+=er*dw/8; eG[pi+width*2]+=eg*dw/8; eB[pi+width*2]+=eb*dw/8; }
                    } else {
                        // ── Floyd-Steinberg dithering (7-3-5-1 / 16) ──
                        const er_w=er*dw, eg_w=eg*dw, eb_w=eb*dw;
                        if (x < width-1) { eR[pi+1]+=er_w*7/16; eG[pi+1]+=eg_w*7/16; eB[pi+1]+=eb_w*7/16; }
                        if (y < height-1) {
                            if (x > 0)       { eR[pi+width-1]+=er_w*3/16; eG[pi+width-1]+=eg_w*3/16; eB[pi+width-1]+=eb_w*3/16; }
                            eR[pi+width]  +=er_w*5/16; eG[pi+width]  +=eg_w*5/16; eB[pi+width]  +=eb_w*5/16;
                            if (x < width-1) { eR[pi+width+1]+=er_w*1/16; eG[pi+width+1]+=eg_w*1/16; eB[pi+width+1]+=eb_w*1/16; }
                        }
                    }
                }
            }
            return indexed;
        }

        function lzwEncode(indices, minCodeSize) {
            const HSIZE=5003, clear=1<<minCodeSize, eoi=clear+1;
            const htab=new Int32Array(HSIZE).fill(-1), ctab=new Uint16Array(HSIZE);
            let nextCode=eoi+1, codeSize=minCodeSize+1, maxCode=(1<<codeSize)-1;
            let curByte=0, curBits=0;
            const acBuf=new Uint8Array(256); let acLen=0;
            const rawOut=[minCodeSize];
            function flushAccum(){if(!acLen)return;rawOut.push(acLen);for(let i=0;i<acLen;i++)rawOut.push(acBuf[i]);acLen=0;}
            function charOut(c){acBuf[acLen++]=c;if(acLen>=254)flushAccum();}
            function writeBits(code){curByte|=code<<curBits;curBits+=codeSize;while(curBits>=8){charOut(curByte&0xff);curByte>>>=8;curBits-=8;}}
            function clearHash(){htab.fill(-1);}
            function resetEncoder(){writeBits(clear);clearHash();nextCode=eoi+1;codeSize=minCodeSize+1;maxCode=(1<<codeSize)-1;}
            clearHash();writeBits(clear);
            let ent=indices[0];
            for(let i=1;i<indices.length;i++){
                const c=indices[i],fcode=(c<<12)+ent;
                let hval=(c<<4)^ent;
                const disp=hval===0?1:HSIZE-hval;
                let found=false;
                while(htab[hval]!==-1){if(htab[hval]===fcode){ent=ctab[hval];found=true;break;}hval-=disp;if(hval<0)hval+=HSIZE;}
                if(!found){writeBits(ent);ent=c;if(nextCode<=0xfff){ctab[hval]=nextCode++;htab[hval]=fcode;if(nextCode>maxCode+1&&codeSize<12){codeSize++;maxCode=(1<<codeSize)-1;}}else{resetEncoder();}}
            }
            writeBits(ent);writeBits(eoi);
            if(curBits>0)charOut(curByte&0xff);
            flushAccum();rawOut.push(0);
            return rawOut;
        }

        function encodeOneFrame(imageData, width, height, quality) {

        }

        async function encodeFrames(imgDataArr, width, height, delayMsOrArr, onProgress, quality) {

            return new Blob([bytes],{type:'image/gif'});
        }

    //  bhState 0 → 1 → 2 → 0 → …
    //  0: Full single-panel (280px)

    const dragHandle = guiEl.querySelector('.gume-hdr');

    dragHandle.addEventListener('mousedown', (e) => {
        if (e.button !== 0) return;
        isDragging  = true; _bhDragDist = 0;
        dragHandle.style.cursor = 'grabbing';
        const r = host.getBoundingClientRect();
        dragOX = e.clientX - r.left; dragOY = e.clientY - r.top;
        e.preventDefault();
    });
    document.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        _bhDragDist += Math.hypot(e.movementX||0, e.movementY||0);
        host.style.right = 'auto';
        host.style.left  = Math.max(0, e.clientX - dragOX) + 'px';
        host.style.top   = Math.max(0, e.clientY - dragOY) + 'px';
    });
    document.addEventListener('mouseup', () => {
        if (isDragging) { isDragging = false; dragHandle.style.cursor = 'grab'; }
    });

        });
    }

    const bhEl = guiEl.querySelector('.bh');

        e.stopPropagation();

        bhState = (bhState + 1) % 3;
        guiEl.classList.toggle('gume-split',  bhState === 1);
        guiEl.classList.toggle('gume-bhonly', bhState === 2);

    function installSpeedGuard(video) {
        if (video._gumeGuard) return;
        video._gumeGuard = true;
        video.addEventListener('ratechange', () => {
            if (_settingRate || isRecording || currentRate < 0) return;
            const expected = currentRate === 0 ? 0.01 : currentRate;
            if (Math.abs(video.playbackRate - expected) > 0.05) {
                _settingRate = true; video.playbackRate = expected; _settingRate = false;
            }
        });
    }

    function applySpeed(speed) {
        currentRate = Math.max(-14, Math.min(14, parseFloat(speed)));
        sv.innerText = currentRate.toFixed(2) + 'x';
        ss.value = currentRate;
        const videos = document.querySelectorAll('video');
        if (currentRate >= 0) {
            if (_reverseMuted) {
                _reverseMuted = false;
                document.querySelectorAll('video').forEach(v => {
                    v.muted = false;
                    if (currentVolume > 100) { const n=gainNodes.get(v); if(n) n.gainNode.gain.value=currentVolume/100; }
                });
            }
            if (reverseRafId) { cancelAnimationFrame(reverseRafId); reverseRafId = reverseLastTs = null; }
            const rate = currentRate === 0 ? 0.01 : currentRate;
            videos.forEach(vid => {
                installSpeedGuard(vid);
                _settingRate=true; vid.playbackRate=rate; _settingRate=false;
                if (!vid.paused && !isRecording) vid.play().catch(()=>{});
            });
            if (IS_YOUTUBE) {
                const ytP = document.querySelector('#movie_player');
                if (ytP && typeof ytP.setPlaybackRate==='function')
                    try { ytP.setPlaybackRate(Math.min(Math.max(rate,0.25),2)); } catch(_){}
            }
        } else {
            if (!_reverseMuted) {
                _reverseMuted = true;
                document.querySelectorAll('video').forEach(v => {
                    v.muted = true;
                    const n=gainNodes.get(v);
                    if(n && audioCtx) n.gainNode.gain.setTargetAtTime(0, audioCtx.currentTime, 0.01);
                });
            }
            videos.forEach(vid => { if(!vid.paused) vid.pause(); });
            if (!reverseRafId) {
                const tick = (ts) => {
                    if (!reverseLastTs) reverseLastTs = ts;
                    const delta = (ts - reverseLastTs) / 1000; reverseLastTs = ts;
                    document.querySelectorAll('video').forEach(v => {
                        v.currentTime = Math.max(0, v.currentTime + currentRate * delta);
                    });
                    reverseRafId = requestAnimationFrame(tick);
                };
                reverseRafId = requestAnimationFrame(tick);
            }
        }
    }

    // §16B · TIMELINE SCRUBBER ENGINE
    //
    //  Creates a drag-and-drop scrubber linked to hidden start/end
    //  inputs.  Timecodes are displayed in HH:MM:SS.ff format.
    //  Both handles support pointer capture for smooth drag-outside-
    //  track behaviour.  Clicking the track snaps the nearest handle.
    // ═══════════════════════════════════════════════════════════════
    function formatTimecode(sec, fps) {
        fps = fps || 30;
        sec = Math.max(0, sec);
        const totalF = Math.floor(sec * fps);
        const frames = totalF % Math.round(fps);
        const s      = Math.floor(sec);
        const hh     = Math.floor(s / 3600);
        const mm     = Math.floor((s % 3600) / 60);
        const ss     = s % 60;
        return `${String(hh).padStart(2,'0')}:${String(mm).padStart(2,'0')}:${String(ss).padStart(2,'0')}.${String(frames).padStart(2,'0')}`;
    }

    function initScrubber(opts) {
        // opts: { trackEl, fillEl, hsEl, heEl, tsEl, teEl, tdEl, stInput, enInput, getFps }
        const { trackEl, fillEl, hsEl, heEl, tsEl, teEl, tdEl, stInput, enInput, getFps } = opts;
        let duration = 60;
        let startFrac = 0, endFrac = 1;
        let dragging = null;

        function update() {
            const sf = Math.max(0, Math.min(endFrac - 1e-6, startFrac));
            const ef = Math.max(sf + 1e-6, Math.min(1, endFrac));
            hsEl.style.left = (sf * 100).toFixed(3) + '%';
            heEl.style.left = (ef * 100).toFixed(3) + '%';
            fillEl.style.left  = (sf * 100).toFixed(3) + '%';
            fillEl.style.width = ((ef - sf) * 100).toFixed(3) + '%';
            const st = sf * duration, en = ef * duration;
            stInput.value = st.toFixed(4);
            enInput.value = en.toFixed(4);
            const fps = getFps ? getFps() : 30;
            tsEl.textContent = formatTimecode(st, fps);
            teEl.textContent = formatTimecode(en, fps);
            tdEl.textContent = '⏱ ' + formatTimecode(en - st, fps);
        }

        function fracFromEvent(e) {
            const rect = trackEl.getBoundingClientRect();
            return Math.max(0, Math.min(1, (e.clientX - rect.left) / rect.width));
        }

        hsEl.addEventListener('pointerdown', e => {
            dragging = 'start'; hsEl.setPointerCapture(e.pointerId); e.stopPropagation();
        });
        heEl.addEventListener('pointerdown', e => {
            dragging = 'end'; heEl.setPointerCapture(e.pointerId); e.stopPropagation();
        });

        const onMove = (e) => {
            if (!dragging) return;
            const f = fracFromEvent(e);
            if (dragging === 'start') startFrac = Math.max(0, Math.min(endFrac - 1e-4, f));
            else endFrac = Math.min(1, Math.max(startFrac + 1e-4, f));
            update();
        };
        const onUp = () => { dragging = null; };

        trackEl.addEventListener('pointermove', onMove);
        trackEl.addEventListener('pointerup', onUp);
        trackEl.addEventListener('pointerleave', e => { if (e.buttons === 0) onUp(); });

        // Click on track body → snap nearest handle
        trackEl.addEventListener('click', e => {
            if (e.target === hsEl || e.target === heEl) return;
            const f = fracFromEvent(e);
            if (Math.abs(f - startFrac) <= Math.abs(f - endFrac))
                startFrac = Math.max(0, Math.min(endFrac - 1e-4, f));
            else
                endFrac = Math.min(1, Math.max(startFrac + 1e-4, f));
            update();
        });

        update();

        return {
            setDuration(dur) {
                duration = Math.max(0.1, dur);
                // Keep existing fractions but clamp
                endFrac = Math.min(endFrac, 1);
                update();
            },
            setRange(s, e) {
                startFrac = Math.max(0, Math.min(1, s / duration));
                endFrac   = Math.max(0, Math.min(1, e / duration));
                update();
            },
        };
    }

    function _abortRender() {

    async function seekToVerified(video, targetTime, frameDur) {
        const alreadyClose = Math.abs(video.currentTime - targetTime) < frameDur * 0.25;
        const _waitForFrame = () => new Promise(resolve => {
            if (typeof video.requestVideoFrameCallback === 'function') {
                video.requestVideoFrameCallback((_now, meta) => resolve(meta ? meta.mediaTime : null));
            } else {
                requestAnimationFrame(() => requestAnimationFrame(() => resolve(null)));
            }
        });
        if (alreadyClose) { await _waitForFrame(); return; }
        const _doSeek = (time) => new Promise((resolve, reject) => {
            const onSeeked = () => { video.removeEventListener('seeked',onSeeked); video.removeEventListener('error',onError); _waitForFrame().then(resolve).catch(reject); };
            const onError  = () => { video.removeEventListener('seeked',onSeeked); video.removeEventListener('error',onError); reject(new Error('seek error at '+time)); };
            video.addEventListener('seeked',onSeeked,{once:true}); video.addEventListener('error',onError,{once:true});
            video.currentTime = time;
        });
        let decodedTime = await _doSeek(targetTime);
        if (decodedTime !== null && Math.abs(decodedTime - targetTime) > frameDur * 0.75)
            decodedTime = await _doSeek(targetTime);
    }

    async function captureFrameSeek(v, ctx, cw, ch, startT, frameInterval, totalFrames, quality, onFrame, onProgress) {
        for (let i = 0; i < totalFrames; i++) {
            if (_gifAbortFlag) return false;
            await seekToVerified(v, startT + i * frameInterval, frameInterval);
            if (_gifAbortFlag) return false;
            ctx.drawImage(v, 0, 0, cw, ch);
            if (i === 0 && _GUME_GIF.isCanvasBlack(ctx, cw, ch)) {
                if (!confirm('Cảnh báo: Frame đầu bị đen (DRM/Hardware Accel).\nTắt Hardware Acceleration và thử lại.\n\nTiếp tục?')) return false;
            }
            const imageData = ctx.getImageData(0, 0, cw, ch);
            onFrame(i, _GUME_GIF.encodeOneFrame(imageData, cw, ch, quality));
            onProgress((i+1)/totalFrames);
            await new Promise(r => setTimeout(r, 0));
        }
        return true;
    }

    async function captureFramePlayback(v, ctx, cw, ch, startT, endT, frameInterval, totalFrames, quality, onFrame, onProgress) {
        if (typeof v.requestVideoFrameCallback !== 'function')
            return captureFrameSeek(v, ctx, cw, ch, startT, frameInterval, totalFrames, quality, onFrame, onProgress);

        return new Promise(resolve => {
            let frameIndex=0, nextTargetTime=startT, lastCaptured=-1, rafCancelled=false;
            const PLAY_RATE = 0.03; // v4.0: slower for better frame coverage (was 0.05)
            const onInitialSeeked = () => {
                v.removeEventListener('seeked', onInitialSeeked);
                if (_gifAbortFlag) { resolve(false); return; }
                _settingRate=true; v.playbackRate=PLAY_RATE; _settingRate=false;
                v.play().catch(()=>{});
                scheduleNextFrame();
            };
            v.addEventListener('seeked', onInitialSeeked, { once:true });
            v.currentTime = startT;

            function scheduleNextFrame() {
                if (rafCancelled || _gifAbortFlag) { resolve(false); return; }
                v.requestVideoFrameCallback(async(_now, meta) => {
                    if (rafCancelled || _gifAbortFlag) { resolve(false); return; }
                    const mediaTime = meta ? meta.mediaTime : v.currentTime;
                    if (mediaTime > endT + frameInterval * 0.5) { v.pause(); resolve(true); return; }
                    if (mediaTime >= nextTargetTime - frameInterval * 0.4
                        && frameIndex < totalFrames
                        && Math.round(mediaTime*1000) !== Math.round(lastCaptured*1000))
                    {
                        lastCaptured = mediaTime;
                        ctx.drawImage(v, 0, 0, cw, ch);
                        if (frameIndex === 0 && _GUME_GIF.isCanvasBlack(ctx, cw, ch)) {
                            rafCancelled=true; v.pause();
                            if (!confirm('Cảnh báo: Frame đầu bị đen.\nTắt Hardware Acceleration.\n\nTiếp tục?')) { resolve(false); return; }
                            rafCancelled=false;
                        }
                        const imageData = ctx.getImageData(0, 0, cw, ch);
                        onFrame(frameIndex, _GUME_GIF.encodeOneFrame(imageData, cw, ch, quality));
                        onProgress((frameIndex+1)/totalFrames);
                        frameIndex++;
                        nextTargetTime = startT + frameIndex * frameInterval;
                        if (frameIndex >= totalFrames) { v.pause(); resolve(true); return; }
                    }
                    await new Promise(r => setTimeout(r, 0));
                    scheduleNextFrame();
                });
            }
        });
    }

    if(document.body) startObserver();
    else document.addEventListener('DOMContentLoaded',startObserver,{once:true});

    // ═══════════════════════════════════════════════════════════════
    // §22 · HEALTH CHECK INTERVAL  (1.5s)
    // ═══════════════════════════════════════════════════════════════
    setInterval(()=>{
        const videos=[...document.querySelectorAll('video')];
        if(videos.length>0){
            if(!guiVisible&&!userHidden) showGUI();
            if(!isRecording&&currentRate>=0){
                const rate=currentRate===0?0.01:currentRate;
                videos.forEach(v=>{
                    if(!v._gumeGuard) installSpeedGuard(v);
                    if(Math.abs(v.playbackRate-rate)>0.08){_settingRate=true;v.playbackRate=rate;_settingRate=false;}
                    if(!_reverseMuted){
                        if(currentVolume<=100){if(Math.abs(v.volume-currentVolume/100)>0.02)v.volume=currentVolume/100;}
                        else{if(v.volume<0.99)v.volume=1.0;const n=gainNodes.get(v);if(n&&audioCtx){const want=currentVolume/100;if(Math.abs(n.gainNode.gain.value-want)>0.05)n.gainNode.gain.setTargetAtTime(want,audioCtx.currentTime,0.02);}}
                    }
                });
            }
        } else {
            if(guiVisible) hideGUI();
            if(espActive) $('gume-esp').click();
        }
    }, 1500);

    // ═══════════════════════════════════════════════════════════════
    // §23 · INITIAL DETECTION
    // ═══════════════════════════════════════════════════════════════
    const _checkInitial=()=>{ if(document.querySelector('video')||document.querySelector('img[src*=".gif" i]')) showGUI(); };
    if(document.readyState==='loading') document.addEventListener('DOMContentLoaded',_checkInitial,{once:true});
    else _checkInitial();

})();
